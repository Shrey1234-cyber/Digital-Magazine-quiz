    <script>
        const quizData = [
            {
                category: "Editorial Design",
                question: "Which web layout approach automatically adapts grid structures across devices?",

                    "Fixed-layout pagination",
                    "Responsive web design",
                    "Static PDF conversion",
                    "Monolithic print grid"
                ],
                correct: 1
            },
            {

                    "Replaces ads with pop-ups"
                ],

            },
            {
                category: "Standards",

                correct: 1
            },

                    "Locked portrait orientation"
                ],
                correct: 0
            },
            {
                category: "Media Ethics",
                question: "Which legal principle allows quoting excerpts for educational review without permission?",
                options: [
                    "Public Domain dedication",

                    "Open Source Licensing"
                ],
                correct: 2
            },
            {
                category: "Monetization",
                question: "Which revenue model balances recurring income with gated premium content?",
                options: [
                    "Banner advertising exclusively",
                    "Tiered subscription paywall model",
                    "One-time life purchase",
                    "Government subsidies"
                ],
                correct: 1
            },
            {
                category: "Workflow",
                question: "What is the primary role of a Content Management System (CMS)?",
                options: [
                    "Physical printing at high speeds",
                    "Streamlining collaborative drafting, review, and web publishing",
                    "Encrypting credit cards",
                    "Generating legal disclaimers"
                ],
                correct: 1
            },
            {
                category: "Accessibility",
                question: "Why is adding descriptive 'alt text' to digital magazine images essential?",
                options: [
                    "Speeds up mobile downloads",
                    "Allows screen-reader users to understand visuals",
                    "Prevents search engine indexing",
                    "Mandated for print typography"
                ],
                correct: 1
            },
            {
                category: "Future Trends",
                question: "How is Artificial Intelligence (AI) influencing modern digital magazines?",
                options: [
                    "Replacing all human journalists",
                    "Assisting with transcription, recommendations, and layout generation",
                    "Eliminating security risks",
                    "Converting articles to printing presses"
                ],
                correct: 1
            }
        ];

        let currentQuestion = 0;
        let userAnswers = {};

        function startQuiz() {
            document.getElementById('view-welcome').classList.add('hidden');
            document.getElementById('view-quiz').classList.remove('hidden');
            renderQuestion();
        }

        function renderQuestion() {
            const q = quizData[currentQuestion];
            document.getElementById('current-q-num').innerText = currentQuestion + 1;
            document.getElementById('question-category').innerText = q.category;
            document.getElementById('question-text').innerText = q.question;

            // Progress bar
            const progress = ((currentQuestion + 1) / quizData.length) * 100;
            document.getElementById('progress-bar').style.width = progress + '%';

            // Options
            const container = document.getElementById('options-container');
            container.innerHTML = '';

            q.options.forEach((opt, idx) => {
                const isSelected = userAnswers[currentQuestion] === idx;
                const btn = document.createElement('button');
                btn.className = `w-full text-left p-3 border transition duration-150 text-xs sm:text-sm flex items-center gap-2.5 ${
                    isSelected 
                        ? 'border-mag-dark bg-mag-dark text-white font-semibold' 
                        : 'border-mag-border bg-white hover:bg-mag-light text-mag-dark'
                }`;
                btn.innerHTML = `
                    <span class="inline-flex items-center justify-center w-5 h-5 rounded-none text-[10px] font-bold border ${isSelected ? 'border-white text-white' : 'border-mag-dark text-mag-dark'} flex-shrink-0">
                        ${String.fromCharCode(65 + idx)}
                    </span>
                    <span>${opt}</span>
                `;
                btn.onclick = () => selectOption(idx);
                container.appendChild(btn);
            });

            // Buttons
            document.getElementById('btn-prev').disabled = currentQuestion === 0;
            const nextBtn = document.getElementById('btn-next');
            if (currentQuestion === quizData.length - 1) {
                nextBtn.innerText = 'Submit Quiz 🎯';
                nextBtn.onclick = submitQuiz;
            } else {
                nextBtn.innerText = 'Next →';
                nextBtn.onclick = nextQuestion;
            }
        }

        function selectOption(optionIndex) {
            userAnswers[currentQuestion] = optionIndex;
            renderQuestion();
        }

        function nextQuestion() {
            if (currentQuestion < quizData.length - 1) {
                currentQuestion++;
                renderQuestion();
            }
        }

        function prevQuestion() {
            if (currentQuestion > 0) {
                currentQuestion--;
                renderQuestion();
            }
        }

        function submitQuiz() {
            let score = 0;
            quizData.forEach((q, idx) => {
                if (userAnswers[idx] === q.correct) {
                    score++;
                }
            });

            document.getElementById('view-quiz').classList.add('hidden');
            document.getElementById('view-result').classList.remove('hidden');
            document.getElementById('final-score').innerText = score;

            let gradeText = "Grade: Needs Revision";
            if (score === 10) gradeText = "Grade: Editorial Master (A+)";
            else if (score >= 8) gradeText = "Grade: Distinction (A)";
            else if (score >= 6) gradeText = "Grade: First Class (B)";

            document.getElementById('score-grade').innerText = gradeText;
        }

        function restartQuiz() {
            currentQuestion = 0;
            userAnswers = {};
            document.getElementById('view-result').classList.add('hidden');
            document.getElementById('view-welcome').classList.remove('hidden');
        }

        function shareQuiz() {
            if (navigator.share) {
                navigator.share({
                    title: 'Digital Magazines Quiz',
                    text: 'Take this Class 12 CBSE English Project quiz on Digital Magazines!',
                    url: window.location.href
                }).catch(() => {});
            } else {
                navigator.clipboard.writeText(window.location.href);
                alert('Quiz link copied to clipboard!');
            }
        }
